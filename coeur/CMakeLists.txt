cmake_minimum_required(VERSION 3.21)
project(g1_coeur C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(G1_JOLT "Adaptateur physique Jolt Physics (MIT), sélectionné par \"physique\": \"jolt\" dans reglages.json" OFF)
option(R88_ENABLE_HAVOK "Adaptateur Havok Physics (nécessite le SDK propriétaire)" OFF)

# Lua 5.4 (scénarios scriptés) : graines internes figées pour le déterminisme ; os et io remplacés par des modules vides
file(GLOB LUA_SRC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/lua/*.c)
add_library(g1_lua STATIC ${LUA_SRC})
target_include_directories(g1_lua PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/lua)
target_compile_definitions(g1_lua PRIVATE "luai_makeseed(L)=0" "l_randomizePivot()=0")

file(GLOB G1_SRC ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
add_library(g1_coeur STATIC ${G1_SRC})
target_include_directories(g1_coeur PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
target_compile_definitions(g1_coeur PUBLIC JSON_NOEXCEPTION SOL_NO_EXCEPTIONS=1)
target_link_libraries(g1_coeur PUBLIC g1_lua)

if(G1_JOLT)
  include(FetchContent)
  set(CROSS_PLATFORM_DETERMINISTIC ON CACHE BOOL "" FORCE)
  set(ENABLE_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
  set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "" FORCE)
  set(TARGET_UNIT_TESTS OFF CACHE BOOL "" FORCE)
  set(TARGET_HELLO_WORLD OFF CACHE BOOL "" FORCE)
  set(TARGET_PERFORMANCE_TEST OFF CACHE BOOL "" FORCE)
  set(TARGET_SAMPLES OFF CACHE BOOL "" FORCE)
  set(TARGET_VIEWER OFF CACHE BOOL "" FORCE)
  set(USE_AVX2 OFF CACHE BOOL "" FORCE)
  set(USE_AVX OFF CACHE BOOL "" FORCE)
  set(USE_F16C OFF CACHE BOOL "" FORCE)
  set(USE_FMADD OFF CACHE BOOL "" FORCE)
  set(USE_LZCNT OFF CACHE BOOL "" FORCE)
  set(USE_TZCNT OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(JoltPhysics GIT_REPOSITORY https://github.com/jrouwe/JoltPhysics GIT_TAG v5.3.0 GIT_SHALLOW TRUE SOURCE_SUBDIR Build)
  FetchContent_MakeAvailable(JoltPhysics)
  target_compile_definitions(g1_coeur PUBLIC G1_JOLT)
  target_link_libraries(g1_coeur PUBLIC Jolt)
endif()
if(R88_ENABLE_HAVOK)
  find_package(Havok REQUIRED)
  target_compile_definitions(g1_coeur PUBLIC R88_USE_HAVOK)
  target_link_libraries(g1_coeur PUBLIC Havok::hkPhysics)
endif()

if(PROJECT_IS_TOP_LEVEL)
  add_executable(g1_sim sim/g1_sim.cpp)
  target_link_libraries(g1_sim g1_coeur)
  add_executable(g1_tests tests/test_main.cpp)
  target_link_libraries(g1_tests g1_coeur)
  enable_testing()
  add_test(NAME verifications COMMAND g1_tests ${CMAKE_CURRENT_SOURCE_DIR}/data)
endif()
