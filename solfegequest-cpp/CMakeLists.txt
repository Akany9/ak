cmake_minimum_required(VERSION 3.16)
project(SolfegeQuest LANGUAGES CXX RC)
# Sous Windows (MSVC ou MinGW) : cmake -B build && cmake --build build --config Release
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(SQ_SOURCES src/main.cpp src/http_server.cpp src/gunzip.cpp)

if(WIN32)
  add_executable(SolfegeQuest WIN32 ${SQ_SOURCES} src/assets_win.cpp res/app.rc)
  target_compile_definitions(SolfegeQuest PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(SolfegeQuest PRIVATE ws2_32 shell32 ole32 uuid)
  if(MSVC)
    target_compile_options(SolfegeQuest PRIVATE /utf-8 /W4)
    target_link_options(SolfegeQuest PRIVATE /ENTRY:wWinMainCRTStartup /MANIFEST:NO)
    set_property(TARGET SolfegeQuest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_link_options(SolfegeQuest PRIVATE -municode -static)
  endif()
else()
  add_executable(solfegequest-test ${SQ_SOURCES} src/assets_posix.cpp)
  target_compile_definitions(solfegequest-test PRIVATE SQ_RES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/res")
  find_package(Threads REQUIRED)
  target_link_libraries(solfegequest-test PRIVATE Threads::Threads)
endif()
